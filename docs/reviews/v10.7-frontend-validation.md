// apps/web/src/hooks/useChainStatus.ts
import { usePublicClient, useBlockNumber } from 'wagmi';

export const useChainStatus = () => {
  const publicClient = usePublicClient();
  const { data: blockNumber } = useBlockNumber({
    watch: true,
    cacheTime: 4_000, // 4 seconds
  });
  
  return {
    rpcBlock: blockNumber ? Number(blockNumber) : null,
    rpcConnected: !!publicClient,
  };
};

## Critical Findings from On-Chain Test

### Expected vs Actual Behavior

| Aspect | Expected | Actual | Status |
|--------|----------|--------|--------|
| Spot sell crosses existing orders | ❌ No | ❌ No | ✅ Correct (by design) |
| New order crosses existing spot sell | ✅ Yes | ✅ Yes | ✅ Correct (V10.7 fix!) |
| Orders placed before spot sell | Don't fill | Don't fill | ✅ Correct |
| Orders placed after spot sell | Fill if overlap | Filled 49% | ✅ Correct |

### Key Insight

**Spot sells are PASSIVE** (makers):
- They don't actively cross existing orders when placed
- They wait for new takers to arrive
- This is standard orderbook design

**Range orders are ACTIVE** (takers):
- They actively search for liquidity when placed
- V10.7 ensures they search the FULL buyer's tick range
- This is what we validated!
